def autocorrelate_and_fit(
        data, cor_options: fcsdc.FCSCorParams, fit_options: dict,
        param: lmfit.Parameters
):
    cor = autocorrelate(
        data, m=cor_options["m"], deltat=cor_options["deltat"],
        normalize=cor_options["normalize"], compress=cor_options["compress"])
    res = lmfit.minimize(
        eq.residual, param, args=(cor[:, 0], cor[:, 1], options)
    )
    fit = eq.equation_(res.params, cor[:, 0], options)
    return cor, res, fit
